\begin{columns}\column{1}\block{}{\sectionrule\vspace{0.6em}\eyebrow{Formal background}}\end{columns}
\begin{columns}
\column{0.35}\block{}{\contributionitem{1 \enspace Causal knowledge bases}{%
    Let $U,V$ partition the atoms into \textbf{background} and \textbf{explainable}
    atoms. A \textcolor{aigblue}{\textbf{Boolean causal model}} is a triple
    $(U,V,K)$; a \textcolor{aigblue}{\textbf{causal knowledge base}} is
    $\Delta=(K,A)$.}{%
    \vspace{0.65em}
    \definitionrow{%
      \(\displaystyle K=\{v\leftrightarrow\varphi_v\mid v\in V\}\)}{%
      Boolean structural equations}{%
      one equation per explainable atom}
    \vspace{0.8em}
    \definitionrow{%
      \(\displaystyle A\subseteq\operatorname{Lit}(U),\quad
      \forall u\in U:\ A\cap\{u,\neg u\}\neq\varnothing\)}{%
      Background assumptions}{%
      at least one literal per background atom}}}
\column{0.33}\block{}{\contributionitem{2 \enspace Induced argumentation frameworks}{%
    Given $\Delta=(K,A)$, the \textcolor{aigblue}{\textbf{induced AF}} is
    $F(\Delta)=(\operatorname{Args}(\Delta),\Rightarrow)$.}{%
    \vspace{0.45em}
    \definitionlead{An \textcolor{aigblue}{\textbf{argument}}
      is a pair $a=(\Phi,\psi)\in\operatorname{Args}(\Delta)$:}
    \vspace{0.2em}
    \definitionrow{%
      {\fontsize{16}{20}\selectfont\(\displaystyle
      \begin{aligned}
        &\Phi\subseteq A,\quad K\cup\Phi\nvdash\bot,\quad K\cup\Phi\vdash\psi,\\[-0.15em]
        &\forall\Psi\subsetneq\Phi:\ K\cup\Psi\nvdash\psi
      \end{aligned}\)}}{%
      Causal Argument}{%
      premises are consistent and minimal for $\psi$}
    \vspace{0.3em}
    \definitionlead{For arguments $a=(\Phi,\psi)$ and $a'=(\Phi',\psi')$, an
      \textcolor{aigblue}{\textbf{attack}} is defined by:}
    \vspace{0.2em}
    \definitionrow{%
      {\fontsize{16}{20}\selectfont\(\displaystyle
      a\Rightarrow a'\iff\exists\phi'\in\Phi':\ \phi'\equiv\neg\psi\)}}{%
      Undercut attack}{%
      $\psi$ negates a premise of $a'$}}}
\column{0.32}\block{}{\contributionitem{3 \enspace Semantics-based conclusions}{%
    Given $F=(\operatorname{Args},\Rightarrow)$, a
    \textcolor{aigblue}{\textbf{stable extension}} is a set
    $E\subseteq\operatorname{Args}$:}{%
    \vspace{0.45em}
    \definitionrow{%
      {\fontsize{16}{20}\selectfont\(\displaystyle
      \begin{aligned}
        &\forall a,b\in E:\ a\nRightarrow b,\\[-0.15em]
        &\forall a\in\operatorname{Args}\setminus E\ \exists b\in E:\ b\Rightarrow a
      \end{aligned}\)}}{%
      Stable extension}{%
      \mbox{conflict-free; attacks all outside arguments}}
    \vspace{1.2em}
    \definitionlead{For an observation $\varphi$ and conclusion $\psi$,
      \textcolor{aigblue}{\textbf{causal entailment}} is defined by:}
    \vspace{0.2em}
    \definitionrow{%
      {\fontsize{16}{20}\selectfont\(\displaystyle
      \begin{aligned}
        \varphi\mid\!\sim_\Delta\psi\iff{}&
        \forall E\in\mathsf{st}(F(K\cup\{\varphi\},A)):\\[-0.15em]
        &\exists(\Phi,\psi)\in E
      \end{aligned}\)}}{%
      Skeptical acceptance}{%
      $\psi$ is supported in every stable extension}}}
